cmake_minimum_required(VERSION 3.18)
project(kuroann LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3")

option(KUROANN_NATIVE "Compile for the CPU of the build machine (-march=native)" ON)
option(KUROANN_PYTHON "Build the Python module (needs pybind11)" OFF)

find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
find_library(URING_LIB uring REQUIRED)

set(KUROANN_FLAGS -Wall -Wextra)
if(KUROANN_NATIVE)
  list(APPEND KUROANN_FLAGS -march=native)
endif()

set(KUROANN_SOURCES src/build.cpp src/graph.cpp src/index.cpp src/predicate.cpp src/search.cpp src/update.cpp)
add_library(kuroann STATIC ${KUROANN_SOURCES})
target_include_directories(kuroann PUBLIC include PRIVATE src)
target_compile_options(kuroann PRIVATE ${KUROANN_FLAGS})
target_link_libraries(kuroann PUBLIC OpenMP::OpenMP_CXX Threads::Threads ${URING_LIB})

add_executable(kuroann_cli tools/kuroann.cpp)
set_target_properties(kuroann_cli PROPERTIES OUTPUT_NAME kuroann)
target_compile_options(kuroann_cli PRIVATE ${KUROANN_FLAGS})
target_link_libraries(kuroann_cli PRIVATE kuroann)

enable_testing()
add_executable(test_index tests/test_index.cpp)
target_compile_options(test_index PRIVATE ${KUROANN_FLAGS})
target_link_libraries(test_index PRIVATE kuroann)
add_test(NAME test_index COMMAND test_index)

if(SKBUILD)
  set(KUROANN_PYTHON ON)
endif()
if(KUROANN_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  # The module is a shared object, so it gets its own position-independent copy of the library.
  add_library(kuroann_pic STATIC ${KUROANN_SOURCES})
  set_target_properties(kuroann_pic PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_include_directories(kuroann_pic PUBLIC include PRIVATE src)
  target_compile_options(kuroann_pic PRIVATE ${KUROANN_FLAGS} -fno-semantic-interposition)
  target_link_libraries(kuroann_pic PUBLIC OpenMP::OpenMP_CXX Threads::Threads ${URING_LIB})
  pybind11_add_module(kuroann_py python/kuroann_module.cpp)
  set_target_properties(kuroann_py PROPERTIES OUTPUT_NAME kuroann)
  target_compile_options(kuroann_py PRIVATE ${KUROANN_FLAGS})
  target_link_libraries(kuroann_py PRIVATE kuroann_pic)
  install(TARGETS kuroann_py LIBRARY DESTINATION .)
endif()
